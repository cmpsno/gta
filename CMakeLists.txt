cmake_minimum_required(VERSION 3.20)
project(PalmDistrict VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_GAME "Build the graphical sandbox (requires raylib)" ON)
include(CTest)
add_library(sandbox_core src/simulation.cpp src/vec3.cpp)
target_include_directories(sandbox_core PUBLIC src)
if(MSVC)
  target_compile_options(sandbox_core PRIVATE /W4)
else()
  target_compile_options(sandbox_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(sandbox_tests tests/simulation_tests.cpp)
  target_link_libraries(sandbox_tests PRIVATE sandbox_core)
  add_test(NAME sandbox_mechanics COMMAND sandbox_tests)
  add_executable(sandbox_npc_tests tests/npc_tests.cpp)
  target_link_libraries(sandbox_npc_tests PRIVATE sandbox_core)
  add_test(NAME sandbox_npc COMMAND sandbox_npc_tests)
  add_executable(sandbox_interaction_tests tests/interaction_tests.cpp)
  target_link_libraries(sandbox_interaction_tests PRIVATE sandbox_core)
  add_test(NAME sandbox_interactions COMMAND sandbox_interaction_tests)
  add_executable(sandbox_bench tests/bench.cpp)
  target_link_libraries(sandbox_bench PRIVATE sandbox_core)
  add_test(NAME sandbox_bench COMMAND sandbox_bench)
  add_executable(sandbox_vec3_tests tests/vec3_tests.cpp)
  target_link_libraries(sandbox_vec3_tests PRIVATE sandbox_core)
  add_test(NAME sandbox_vec3 COMMAND sandbox_vec3_tests)
  add_executable(sandbox_floor_tests tests/floor_tests.cpp)
  target_link_libraries(sandbox_floor_tests PRIVATE sandbox_core)
  add_test(NAME sandbox_floor COMMAND sandbox_floor_tests)
endif()
if(BUILD_GAME)
  include(FetchContent)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(CUSTOMIZE_BUILD ON CACHE BOOL "" FORCE)
  set(SUPPORT_MODULE_RAUDIO OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(raylib
    GIT_REPOSITORY https://github.com/raysan5/raylib.git
    GIT_TAG c1ab645ca298a2801097931d1079b10ff7eb9df8)
  FetchContent_MakeAvailable(raylib)
  add_executable(palm_district src/main.cpp src/renderer.cpp)
  target_link_libraries(palm_district PRIVATE sandbox_core raylib)
  # Keep upstream C header warnings separate from warnings in our C++ sources.
  target_include_directories(palm_district SYSTEM PRIVATE ${raylib_SOURCE_DIR}/src)
  if(MSVC)
    target_compile_options(palm_district PRIVATE /W4)
  else()
    target_compile_options(palm_district PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(MINGW)
    target_link_options(palm_district PRIVATE -static -static-libgcc -static-libstdc++)
    if(BUILD_TESTING)
      target_link_options(sandbox_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
  endif()
endif()
